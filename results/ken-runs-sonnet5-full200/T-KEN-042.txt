from datetime import datetime, timedelta

def is_open(state):
    if not state['open']:
        return False
    opened_at = datetime.fromisoformat(state['opened_at'])
    if datetime.utcnow() >= opened_at + timedelta(seconds=state['cooldown_sec']):
        return False  # cooldown expired, breaker should close (or go half-open)
