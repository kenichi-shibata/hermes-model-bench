```python
import re

PART_SUFFIX_RE = re.compile(
    r"""
    [\s\-:—–]*                                   # optional separator
    [\(\[]?                                      # optional opening bracket
    (?:part|pt\.?|vol\.?|volume|episode|ep\.?)   # 'part' keyword
    [\s\.]*
    (?:\d+|[ivxlcdm]+)                           # number or roman numeral
    [\)\]]?                                      # optional closing bracket
    \s*$
    """,
    re.IGNORECASE | re.VERBOSE,
)


def base_title(title):
    return PART_SUFFIX_RE.sub('', title or '').strip()


def group_scenes(scenes):
    groups = {}
    for scene in scenes:
        key = scene.get('movie_id')
        if key is None:
            key = base_title(scene.get('title'))
        groups.setdefault(key, []).append(scene)
    return groups


def assign_movie_ids(scenes):
    existing = [s['movie_id'] for s in scenes if isinstance(s.get('movie_id'), int)]
    next_id = max(existing) + 1 if existing else 1
    ids_by_base = {}
    for scene in scenes:
        if scene.get('movie_id') is not None:
            continue
        base = base_title(scene.get('title'))
        if base not in ids_by_base:
            ids_by_base[base] = next_id
            next_id += 1
        scene['movie_id'] = ids_by_base[base]
    return scenes


def find_ungrouped(scenes):
    return [s for s in scenes if s.get('movie_id') is None]
```
