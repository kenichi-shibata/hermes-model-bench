import json, queue, threading, time
from collections import Counter
from retry_api_mock import retry

backlog = json.load(open('errored_backlog.json'))
errored = [it for it in backlog if it.get('status') == 'error']

def run(workers):
    q = queue.Queue()
    for it in errored: q.put(it)
    results, lock, stats = [], threading.Lock(), Counter()
    def worker():
        while True:
            try: item = q.get_nowait()
            except queue.Empty: return
            while True:
                try:
                    res = retry(item['id'])
                    with lock: results.append(res); stats['resolved'] += 1
                    break
                except Exception as e:
                    if '429' in str(e):
                        with lock: stats['429s'] += 1
                        time.sleep(0.001)
                    else:
                        with lock: stats['failed'] += 1
                        break
            q.task_done()
    ts = [threading.Thread(target=worker) for _ in range(workers)]
    t0 = time.time()
    [t.start() for t in ts]; [t.join() for t in ts]
    return time.time()-t0, stats, len(results)

for w in (5,):
    el, st, n = run(w)
    print(f'workers={w}: {el:.3f}s resolved={st[\"resolved\"]} failed={st[\"failed\"]} 429s={st[\"429s\"]} results={n}')
