import unittest
from unittest.mock import patch, MagicMock
import requests
from requests.exceptions import ConnectionError, Timeout

TRANSIENT_STATUS_CODES = {429, 500, 502, 503, 504}
TRANSIENT_EXCEPTIONS = (
    requests.exceptions.ConnectionError,
    requests.exceptions.Timeout,
    ConnectionError,
    TimeoutError,
)

def call_flaky_api(url=None, max_retries=3, backoff_factor=1, **kwargs):
    if url is None:
        url = globals().get('url', 'https://api.example.com/data')
    if 'timeout' not in kwargs:
        kwargs['timeout'] = 10

    for attempt in range(max_retries):
        try:
            response = requests.get(url, **kwargs)
            if response.status_code in TRANSIENT_STATUS_CODES and attempt < max_retries - 1:
                sleep_time = backoff_factor * (2 ** attempt)
                import time
                time.sleep(sleep_time)
                continue
            return response
        except TRANSIENT_EXCEPTIONS as exc:
            if attempt < max_retries - 1:
                sleep_time = backoff_factor * (2 ** attempt)
                import time
                time.sleep(sleep_time)
            else:
                raise exc

# Test 1: 404 does not retry
with patch('requests.get') as mock_get:
    mock_resp = MagicMock()
    mock_resp.status_code = 404
    mock_get.return_value = mock_resp
    res = call_flaky_api('http://test.com', backoff_factor=0)
    assert mock_get.call_count == 1
    assert res.status_code == 404
    print('Test 1 (404 no retry) passed!')

    assert res.status_code == 200
