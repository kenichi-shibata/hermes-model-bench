import datetime
from datetime import timezone

def parse_timestamp(val):
    if val is None:
        return None
    if isinstance(val, datetime.datetime):
        if val.tzinfo is None:
            return val.replace(tzinfo=timezone.utc)
        return val.astimezone(timezone.utc)
    if isinstance(val, datetime.date):
        return datetime.datetime(val.year, val.month, val.day, tzinfo=timezone.utc)
    if isinstance(val, (int, float)):
        # Handle ms vs s
        if val > 1e11: # likely milliseconds
            val = val / 1000.0
        return datetime.datetime.fromtimestamp(val, tz=timezone.utc)
    if isinstance(val, str):
        val = val.strip()
        # Handle 'Z' or ISO
        try:
            dt = datetime.datetime.fromisoformat(val)
            if dt.tzinfo is None:
                return dt.replace(tzinfo=timezone.utc)
            return dt.astimezone(timezone.utc)
        except ValueError:
            for fmt in ('%Y-%m-%d %H:%M:%S', '%Y/%m/%d %H:%M:%S', '%Y-%m-%d'):
                try:
                    dt = datetime.datetime.strptime(val, fmt)
                    return dt.replace(tzinfo=timezone.utc)
                except ValueError:
                    pass
    return None

print(parse_timestamp('2026-08-15 12:00:00'))
print(parse_timestamp('2026-08-15T12:00:00Z'))
print(parse_timestamp(1700000000))
print(parse_timestamp(1700000000000))
