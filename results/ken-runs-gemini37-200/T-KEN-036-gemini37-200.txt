ROMAN_MAP = {"i": 1, "v": 5, "x": 10, "l": 50, "c": 100, "d": 500, "m": 1000}

def roman_to_int(s):
    s = s.lower().strip()
    if not all(c in ROMAN_MAP for c in s):
        return None
    total = 0
    prev = 0
    for char in reversed(s):
        val = ROMAN_MAP[char]
        if val < prev:
            total -= val
        else:
            total += val
            prev = val
    return total

def parse_num_or_roman(val_str):
    if not val_str:
        return None
    val_str = val_str.strip()
    if val_str.isdigit():

    r = roman_to_int(val_str)

    return cleaned, None

def group_by_movie(scenes, assign_ids=True, start_id=1):
    existing_ids = [
        s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)
        for s in scenes
    ]
    existing_numeric_ids = [i for i in existing_ids if isinstance(i, int)]
    next_id = max(existing_numeric_ids, default=start_id - 1) + 1 if existing_numeric_ids else start_id

    groups = {}
    for s in scenes:
        title = extract_movie_title(s)
        if title not in groups:
            groups[title] = []
        groups[title].append(s)

        # sort by part number if available, then by id
        group_scenes.sort(key=lambda x: (extract_part_number(x) or 0, x.get("id", 0) if isinstance(x, dict) else getattr(x, "id", 0)))
        
        existing_m_id = next(
            (
                s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)
                for s in group_scenes
                if (s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)) is not None
            ),
            None,
        )
        if assign_ids:
            if existing_m_id is not None:

            else:
                m_id = next_id
                next_id += 1
            for s in group_scenes:
                if isinstance(s, dict):

                        s["movie_id"] = m_id
                else:
                    if getattr(s, "movie_id", None) is None:
                        setattr(s, "movie_id", m_id)

    return groups

    # "(1 of 6)", "[1 of 6]", "(1/6)"
    r"[\s\-_.,:]*[\(\[]\s*([0-9]+)\s*(?:of|/)\s*\d+\s*[\)\]].*$",
    # "(1)", "[1]"
    r"[\s\-_.,:]*[\(\[]\s*([0-9]+|[ivxldcm]+)\s*[\)\]].*$",
    # " - Part 1", " Part 1", " Pt. 1", " CD 1", " Disc 1", " Scene 1", " Part 1 of 6", " Part IV"
    r"[\s\-_.,:]+(?:part|pt|scene|cd|disc|disk|act)[\s\-_.:#]*([0-9]+|[ivxldcm]+)(?:\s*(?:of|/)\s*\d+)?(?:\b|[\s\-_.:]).*$",
    # " 1 of 6", " - 1 of 6"
    r"[\s\-_.,:]+([0-9]+)\s*(?:of|/)\s*\d+.*$",
    # " - 1", " - 01", "_01" at end
    r"[\s\-_.,:]+0*([0-9]+)\s*$",
]

def extract_movie_title(scene_or_title):
    if isinstance(scene_or_title, dict):
        title = scene_or_title.get("title", "")
    elif hasattr(scene_or_title, "title"):
        title = scene_or_title.title
    else:
        title = str(scene_or_title) if scene_or_title is not None else ""
    cleaned = title.strip()
    for pattern in PART_PATTERNS:
        match = re.search(pattern, cleaned, re.IGNORECASE)
        if match:
            base = cleaned[:match.start()].strip()
            base = re.sub(r"[\s\-_.,:]+$", "", base).strip()
            if base:
                return base
    return cleaned

def extract_part_number(scene_or_title):
    if isinstance(scene_or_title, dict):
        title = scene_or_title.get("title", "")
    elif hasattr(scene_or_title, "title"):
        title = scene_or_title.title
    else:
        title = str(scene_or_title) if scene_or_title is not None else ""
    cleaned = title.strip()
    for pattern in PART_PATTERNS:
        match = re.search(pattern, cleaned, re.IGNORECASE)
        if match:
            part_str = match.group(1) if match.lastindex and match.lastindex >= 1 else None
            return parse_num_or_roman(part_str) if part_str else None
    return None

def find_ungrouped(scenes):
    return [s for s in scenes if (s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)) is None]

def group_by_movie(scenes, assign_ids=True, start_id=1):
    existing_ids = [
        s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)
        for s in scenes
    ]
    existing_numeric_ids = [i for i in existing_ids if isinstance(i, int)]
    next_id = max(existing_numeric_ids, default=start_id - 1) + 1 if existing_numeric_ids else start_id

    groups = {}
    for s in scenes:
        title = extract_movie_title(s)
        if title not in groups:
            groups[title] = []
        groups[title].append(s)

    for movie_title, group_scenes in groups.items():
        group_scenes.sort(key=lambda x: (extract_part_number(x) or 0, x.get("id", 0) if isinstance(x, dict) else getattr(x, "id", 0)))
        
        existing_m_id = next(
            (
                s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)
                for s in group_scenes
                if (s.get("movie_id") if isinstance(s, dict) else getattr(s, "movie_id", None)) is not None
            ),
            None,
        )
        if assign_ids:
            if existing_m_id is not None:
                m_id = existing_m_id
            else:
                m_id = next_id
                next_id += 1
            for s in group_scenes:
                if isinstance(s, dict):
                    if s.get("movie_id") is None:
                        s["movie_id"] = m_id
                else:
                    if getattr(s, "movie_id", None) is None:
                        setattr(s, "movie_id", m_id)

    return groups
