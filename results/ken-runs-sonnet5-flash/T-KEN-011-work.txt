import { useState } from "react";
import { Link } from "react-router-dom";

const PAGE_SIZE = 60;

export default function PerformersGrid({ performers = [] }) {
  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);

  if (performers.length === 0) {
    return <div>No performers found</div>;
  }

  const hasMore = visibleCount < performers.length;

  return (
    <div>
      <div>
        {performers.slice(0, visibleCount).map(p => (
          <Link key={p.id} to={`/performers/${p.id}`}>
            {p.name}
          </Link>
        ))}
      </div>
      {hasMore && (
        <button onClick={() => setVisibleCount(count => Math.min(count + PAGE_SIZE, performers.length))}>
          Load more
        </button>
      )}
    </div>
  );
}
